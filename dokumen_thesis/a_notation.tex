\normalfont\LARGE\textbf{LIST OF NOTATIONS}
\end{center}
\begin{longtable}{|p{2.5cm}|p{11.5cm}|}
	\hline
	\rowcolor{orange!25}\textbf{Symbols} & \textbf{Definition}\\
	\hline \endhead \hline
	$C$ & The cover-image (citra penampung) \\ \hline
	$S$ & The stego-image (citra hasil penyisipan) \\ \hline
	$M \times N$ & The dimensions of the image (width $\times$ height) in pixels \\ \hline
	$P$ & The secret message or payload to be embedded \\ \hline
	$b_k$ & The $k$-th bit of the secret message $P$ \\ \hline
	$C(i,j)$ & The pixel value at coordinate $(i,j)$ of the cover-image \\ \hline
	$S(i,j)$ & The pixel value at coordinate $(i,j)$ of the stego-image \\ \hline
	$C_R, C_G, C_B$ & The Red, Green, and Blue channels of a cover-image \\ \hline
	$L$ & Number of Least Significant Bits (LSB) used for embedding \\ \hline
	$bpp$ & Bits Per Pixel, a measure of embedding capacity \\ \hline
	$MAX_I$ & Maximum possible pixel intensity value (e.g., 255 for an 8-bit image) \\ \hline
	$MSE$ & Mean Squared Error \\ \hline
	$PSNR$ & Peak Signal-to-Noise Ratio, in decibels (dB) \\ \hline
	$\Sigma$ & Summation operator \\ \hline
	$\oplus$ & The Exclusive OR (XOR) operator \\ \hline
	$E(\cdot)$ & The embedding function or algorithm \\ \hline
	$D(\cdot)$ & The extraction or decoding function \\ \hline
\end{longtable}
